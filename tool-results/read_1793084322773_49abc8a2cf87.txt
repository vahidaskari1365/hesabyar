   123→/** تبدیل رشته فارسی/لاتین عدد به number (برای جمع ستون‌ها) */
   124→function parseFa(v: string | number | null | undefined): number {
   125→  if (typeof v === "number") return v;
   126→  if (!v) return 0;
   127→  const n = Number(
   128→    v
   129→      .replace(/[۰-۹]/g, (d) => String("۰۱۲۳۴۵۶۷۸۹".indexOf(d)))
   130→      .replace(/[^\d.-]/g, "")
   131→  );
   132→  return Number.isNaN(n) ? 0 : n;
   133→}
   134→
   135→function csvEscape(v: string | number): string {
   136→  const s = String(v ?? "");
   137→  return /[",\n\r]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
   138→}
   139→
   140→function downloadCsv(data: ReportData) {
   141→  const lines: string[] = [data.columns.map(csvEscape).join(",")];
   142→  for (const row of data.rows) lines.push(row.map(csvEscape).join(","));
   143→  if (data.totals) lines.push(data.totals.map(csvEscape).join(","));
   144→  const blob = new Blob(["\uFEFF" + lines.join("\r\n")], { type: "text/csv;charset=utf-8;" });
   145→  const url = URL.createObjectURL(blob);
   146→  const a = document.createElement("a");
   147→  a.href = url;
   148→  a.download = `${data.title || "گزارش"}.csv`;
   149→  document.body.appendChild(a);
   150→  a.click();
   151→  a.remove();
   152→  URL.revokeObjectURL(url);
   153→  toast.success("خروجی CSV با موفقیت دانلود شد");
   154→}

   164→function ReportChart({ chart }: { chart: NonNullable<ReportData["chart"]> }) {
   165→  if (!chart.data?.length) {
   166→    return (
   167→      <div className="flex h-full items-center justify-center text-sm text-muted-foreground">
   168→        داده‌ای برای نمایش نمودار وجود ندارد
   169→      </div>
   170→    );
   171→  }
   172→
   173→  if (chart.kind === "pie") {
   174→    return (
   175→      <ResponsiveContainer width="100%" height="100%">
   176→        <PieChart>
   177→          <Pie
   178→            data={chart.data}
   179→            dataKey="value"
   180→            nameKey="name"
   181→            innerRadius="52%"
   182→            outerRadius="82%"
   183→            paddingAngle={2}
   184→            stroke="var(--card)"
   185→            strokeWidth={2}
   186→          >
   187→            {chart.data.map((d, i) => (
   188→              <Cell key={`${d.name}-${i}`} fill={PIE_COLORS[i % PIE_COLORS.length]} />
   189→            ))}
   190→          </Pie>
   191→          <RTooltip {...tooltipStyle} formatter={(v) => faCompact(Number(v))} />
   192→          <Legend wrapperStyle={{ fontSize: 11 }} iconSize={9} />
   193→        </PieChart>
   194→      </ResponsiveContainer>
   195→    );
   196→  }
   197→
   198→  if (chart.kind === "line") {
   199→    return (
   200→      <ResponsiveContainer width="100%" height="100%">
   201→        <LineChart data={chart.data} margin={{ top: 8, right: 16, left: 0, bottom: 0 }}>
   202→          <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="var(--border)" />
   203→          <XAxis dataKey="name" tick={axisTick} tickLine={false} axisLine={{ stroke: "var(--border)" }} interval={0} />
   204→          <YAxis tick={axisTick} tickLine={false} axisLine={false} tickFormatter={compactTick} width={48} />

   207→            type="monotone"
   208→            dataKey="value"
   209→            name="خالص گردش"
   210→            stroke={CHART_COLORS[0]}
   211→            strokeWidth={2.5}
   212→            dot={{ r: 3, strokeWidth: 0, fill: CHART_COLORS[0] }}
   213→            activeDot={{ r: 5 }}
   214→          />
   215→        </LineChart>
   216→      </ResponsiveContainer>
   217→    );
   218→  }
   219→
   220→  // bar — اگر نام‌ها بلندند چیدمان افقی (ستون‌های افقی با برچسب راست)
   221→  const hasExtra = chart.data.some((d) => typeof d.extra === "number");
   222→  const longNames = chart.data.some((d) => d.name.length > 10);
   223→
   224→  if (longNames) {
   225→    return (
   226→      <ResponsiveContainer width="100%" height="100%">
   227→        <BarChart data={chart.data} layout="vertical" margin={{ top: 4, right: 16, left: 8, bottom: 0 }}>
   228→          <CartesianGrid strokeDasharray="3 3" horizontal={false} stroke="var(--border)" />
   229→          <XAxis type="number" tick={axisTick} tickLine={false} axisLine={false} tickFormatter={compactTick} />
   230→          <YAxis
   231→            type="category"
   232→            dataKey="name"
   233→            width={140}
   234→            tick={axisTick}
   235→            tickLine={false}
   236→            axisLine={false}
   237→            orientation="right"
   238→          />
   239→          <RTooltip {...tooltipStyle} formatter={(v) => faNumber(Number(v))} cursor={{ fill: "var(--accent)", fillOpacity: 0.35 }} />
   240→          {hasExtra ? (
   241→            <>
   242→              <Bar dataKey="value" name="بدهکار" fill={CHART_COLORS[0]} radius={[0, 4, 4, 0]} barSize={9} />
   243→              <Bar dataKey="extra" name="بستانکار" fill={CHART_COLORS[1]} radius={[0, 4, 4, 0]} barSize={9} />
   244→            </>
   245→          ) : (
   246→            <Bar dataKey="value" name="مبلغ" fill={CHART_COLORS[0]} radius={[0, 4, 4, 0]} barSize={14} />
   247→          )}
   248→          {hasExtra && <Legend wrapperStyle={{ fontSize: 11 }} iconSize={9} />}
   249→        </BarChart>
   250→      </ResponsiveContainer>
   251→    );
   252→  }
   253→
   254→  return (
   255→    <ResponsiveContainer width="100%" height="100%">
   256→      <BarChart data={chart.data} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
   257→        <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="var(--border)" />
   258→        <XAxis dataKey="name" tick={axisTick} tickLine={false} axisLine={{ stroke: "var(--border)" }} interval={0} />
   259→        <YAxis tick={axisTick} tickLine={false} axisLine={false} tickFormatter={compactTick} width={48} />
   260→        <RTooltip {...tooltipStyle} formatter={(v) => faNumber(Number(v))} cursor={{ fill: "var(--accent)", fillOpacity: 0.35 }} />
   261→        <Bar dataKey="value" name="مبلغ" fill={CHART_COLORS[0]} radius={[4, 4, 0, 0]} barSize={26} />
   262→      </BarChart>
   263→    </ResponsiveContainer>
   264→  );
   265→}
